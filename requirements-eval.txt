# Dependencies for ragas_eval_live.py ONLY - kept separate from
# requirements.txt on purpose.
#
# Streamlit Community Cloud installs requirements.txt into a ~1GB container.
# The packages below pull in torch + transformers + datasets (~2GB), which
# would break the deployment. ragas_eval_live.py is deliberately decoupled
# from rag_engine.py (it drives the deployed UI as a black box), so it needs
# none of the app's own runtime deps - no streamlit, chromadb, or groq here.
#
# Installed only by .github/workflows/ragas-live.yml.

ragas
datasets
pandas
python-dotenv

# RAGAS judge LLM (Groq) + the BAAI/bge-base-en-v1.5 embeddings the metrics
# score context similarity with. sentence-transformers is listed explicitly
# because langchain-huggingface treats it as an optional extra, but
# HuggingFaceEmbeddings won't import without it.
langchain-groq
langchain-huggingface
sentence-transformers

# Browser automation against the deployed Streamlit UI. The workflow also
# runs `playwright install --with-deps chromium` - pip alone doesn't fetch
# the browser binary.
playwright

# Left unpinned deliberately: ragas ships breaking metric-API changes between
# minor versions, and pinning to a version I haven't verified green would be
# a guess. After your first successful run, pin them for reproducibility -
# `pip freeze | grep -Ei 'ragas|datasets|langchain|sentence-transformers|playwright'`
# from the workflow log gives you the exact known-good set.
